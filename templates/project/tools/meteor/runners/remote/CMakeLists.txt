cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(qmq_remote_harness LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(meteor_hardware_probe hardware_probe.cpp)
set(METEOR_CANN_LINUX_PATH "")
if(DEFINED ASCEND_CANN_PACKAGE_LINUX_PATH)
    set(METEOR_CANN_LINUX_PATH "${ASCEND_CANN_PACKAGE_LINUX_PATH}")
elseif(DEFINED ENV{ASCEND_HOME_PATH} AND EXISTS "$ENV{ASCEND_HOME_PATH}/include/acl/acl.h")
    set(METEOR_CANN_LINUX_PATH "$ENV{ASCEND_HOME_PATH}")
elseif(DEFINED ENV{ASCEND_HOME_PATH} AND EXISTS "$ENV{ASCEND_HOME_PATH}/aarch64-linux/include/acl/acl.h")
    set(METEOR_CANN_LINUX_PATH "$ENV{ASCEND_HOME_PATH}/aarch64-linux")
elseif(DEFINED ENV{ASCEND_TOOLKIT_HOME} AND EXISTS "$ENV{ASCEND_TOOLKIT_HOME}/include/acl/acl.h")
    set(METEOR_CANN_LINUX_PATH "$ENV{ASCEND_TOOLKIT_HOME}")
elseif(DEFINED ENV{ASCEND_TOOLKIT_HOME} AND EXISTS "$ENV{ASCEND_TOOLKIT_HOME}/aarch64-linux/include/acl/acl.h")
    set(METEOR_CANN_LINUX_PATH "$ENV{ASCEND_TOOLKIT_HOME}/aarch64-linux")
endif()
if(NOT "${METEOR_CANN_LINUX_PATH}" STREQUAL "")
    target_include_directories(meteor_hardware_probe PRIVATE "${METEOR_CANN_LINUX_PATH}/asc/include" "${METEOR_CANN_LINUX_PATH}/include")
    target_link_directories(meteor_hardware_probe PRIVATE "${METEOR_CANN_LINUX_PATH}/lib64")
endif()
target_link_libraries(meteor_hardware_probe PRIVATE tiling_api platform unified_dlog dl pthread ascendcl)

set(NPU_ARCH "" CACHE STRING "Architecture confirmed on the target device")
set(QMQ_KERNEL_PATH "" CACHE FILEPATH "Frozen kernel source to include verbatim")
if(NOT "${QMQ_KERNEL_PATH}" STREQUAL "")
    get_filename_component(QMQ_KERNEL_PATH "${QMQ_KERNEL_PATH}" ABSOLUTE)
    if(NOT EXISTS "${QMQ_KERNEL_PATH}")
        message(FATAL_ERROR "Set QMQ_KERNEL_PATH to an existing frozen kernel.asc")
    endif()

    if("${NPU_ARCH}" STREQUAL "")
        message(FATAL_ERROR "Set NPU_ARCH from verified hardware/profile data")
    endif()

    add_executable(qmq_remote_main main.asc)
    target_compile_definitions(qmq_remote_main PRIVATE QMQ_KERNEL_HEADER="${QMQ_KERNEL_PATH}")
    target_compile_options(qmq_remote_main PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${NPU_ARCH}>)
    target_link_libraries(qmq_remote_main PRIVATE tiling_api register platform unified_dlog dl m graph_base)
endif()

if(NOT "${NPU_ARCH}" STREQUAL "")
    add_executable(meteor_hardware_add hardware_add_host.asc)
    target_compile_options(meteor_hardware_add PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${NPU_ARCH}>)
    target_link_libraries(meteor_hardware_add PRIVATE tiling_api register platform unified_dlog dl pthread m graph_base ascendcl)
endif()
